nom_send_gold = {
	type = situation
	potential = {
	}
	allow = {
		scope:actor ?= {
			gold >= 10
		}
	}

	automation_tick = never
	automation_tick_frequency = 12
	cooldown = { type = needs_of_the_mother_shared_cooldown months = 8 }

	price = price:nom_send_gold

	select_trigger = {
		looking_for_a = situation
		target_flag = recipient
		name = "choose_situation"
		column = { data = name }
		visible = {
			situation:needs_of_the_mother = this
			situation_is_active = yes
		}
	}

	effect = {
		situation:needs_of_the_mother = {
			change_variable = {
				name = nom_support
				add = 10
			}
			clamp_variable = {
				name = nom_support
				max = 100
				min = 0
			}
			needs_of_the_mother_update_milestones = yes
		}
	}

	ai_will_do = {
		add = 25
	}
}
