            }

        }
    }

    option = {
        name = dusk_of_the_old_world.1.a

            limit = {
                is_port = yes

                50 = {
                    add_pop = {culture = scope:target_culture religion = root.religion type = pop_type:peasants size = 0.4}

                    add_pop = {culture = scope:target_culture religion = root.religion type = pop_type:peasants size = 0.8}
                    root = { change_variable = { name = dusk_of_the_old_world_migrated_pops add = 800 } }
                }
                10 = {
                    add_pop = {culture = scope:target_culture religion = root.religion type = pop_type:peasants size = 1.6}
                    root = { change_variable = { name = dusk_of_the_old_world_migrated_pops add = 1600 } }

nw_dusk_of_the_old_world.2 = {
    type = country_event
    title = nw_dusk_of_the_old_world.2.title
    desc = nw_dusk_of_the_old_world.2.desc

    outcome = neutral

    trigger = {
        any_owned_location = {
            any_pop = {
                OR = {
                    NOT = { culture = root.culture }
                    NOT = { religion = root.religion }
                }
            }
        }
    }

    illustration_tags = {
		10 = exterior
	}

    option = {
        name = dusk_of_the_old_world.2.a

        custom_tooltip = {
            text = nw_dusk_assimilation_tt
            random_owned_location = {
                limit = {
                    any_pop = {
                        OR = {
                            NOT = { culture = root.culture }
                            NOT = { religion = root.religion }
                        }
                    }
                }
                every_pop = {
                    limit = {
                        OR = {
                            NOT = { culture = root.culture }
                            NOT = { religion = root.religion }
                        }
                    }
                    split_pop = {
                        fraction = 0.2
                        religion = root.religion
                        culture = root.culture
                    }
                }
            }
        }
    }
}

nw_dusk_of_the_old_world.4 = {
    type = country_event
    title = nw_dusk_of_the_old_world.4.title
    desc = nw_dusk_of_the_old_world.4.desc

    outcome = neutral

    illustration_tags = {
		10 = interior
	}

    immediate = {
        random_list = {
            10 = { set_variable = { name = nw_dusk_merchant_goods_amount value = 5 } }
            10 = { set_variable = { name = nw_dusk_merchant_goods_amount value = 6 } }
            10 = { set_variable = { name = nw_dusk_merchant_goods_amount value = 7 } }
            10 = { set_variable = { name = nw_dusk_merchant_goods_amount value = 8 } }
            10 = { set_variable = { name = nw_dusk_merchant_goods_amount value = 9 } }
            10 = { set_variable = { name = nw_dusk_merchant_goods_amount value = 10 } }
        }
        set_variable = {
            name = nw_dusk_merchant_goods_cost
            value = var:nw_dusk_merchant_goods_amount
        }
        random_list = {
            10 = { set_variable = { name = nw_dusk_merchant_goods_type value = 1 } }
            10 = { set_variable = { name = nw_dusk_merchant_goods_type value = 2 } }
            10 = { set_variable = { name = nw_dusk_merchant_goods_type value = 3 } }
            10 = { set_variable = { name = nw_dusk_merchant_goods_type value = 4 } }
        }
    }

    option = {
        name = dusk_of_the_old_world.4.a

        add_gold = { value = var:nw_dusk_merchant_goods_cost multiply = -1 }

        if = {
            limit = { var:nw_dusk_merchant_goods_type = 1 }
            capital.market ?= { add_goods_supply = { goods = goods:cloth amount = root.var:nw_dusk_merchant_goods_amount } }
        }
        else_if = {
            limit = { var:nw_dusk_merchant_goods_type = 2 }
            capital.market ?= { add_goods_supply = { goods = goods:wool amount = root.var:nw_dusk_merchant_goods_amount } }
        }
        else_if = {
            limit = { var:nw_dusk_merchant_goods_type = 3 }
            capital.market ?= { add_goods_supply = { goods = goods:silver amount = root.var:nw_dusk_merchant_goods_amount } }
        }
        else = {
            capital.market ?= { add_goods_supply = { goods = goods:iron amount = root.var:nw_dusk_merchant_goods_amount } }
        }

        remove_variable = nw_dusk_merchant_goods_amount
        remove_variable = nw_dusk_merchant_goods_cost
        remove_variable = nw_dusk_merchant_goods_type
    }

    option = {
        name = dusk_of_the_old_world.4.b
        remove_variable = nw_dusk_merchant_goods_amount
        remove_variable = nw_dusk_merchant_goods_cost
        remove_variable = nw_dusk_merchant_goods_type
    }
}
