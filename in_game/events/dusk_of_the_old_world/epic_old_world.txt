nw_epic_old_world.1 = {
    type = country_event
    title = nw_epic_old_world.1.title
    desc = nw_epic_old_world.1.desc

    outcome = neutral

    trigger = {
        has_variable = old_world_colony
        is_situation_active = situation:dusk_of_the_old_world
    }

    illustration_tags = {
        10 = interior
    }

    option = {
        name = nw_epic_old_world.1.a
        add_country_modifier = { modifier = nw_epic_old_world_uneasy_markets years = 2 mode = add_and_extend }
        trigger_event_non_silently = { id = nw_epic_old_world.2 months = 24 }
    }

    option = {
        name = nw_epic_old_world.1.b
        add_gold = -25
        add_country_modifier = { modifier = nw_epic_old_world_informed_merchants years = 3 mode = add_and_extend }
        set_variable = nw_epic_old_world_investigated
        trigger_event_non_silently = { id = nw_epic_old_world.2 months = 24 }
    }
}
